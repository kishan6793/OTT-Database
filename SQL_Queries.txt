=> Query :

1. Find the top 5 most-watched movies in 2024.
=> SELECT c.Title, s.Total_View
   FROM (Content natural join movies) c
   JOIN Statistics s ON c.Content_ID = s.Content_ID
   WHERE EXTRACT(YEAR FROM c.Release_Date) = 2024
   ORDER BY s.Total_View DESC;

2. List premium users who streamed more than 5 hours in the last month.
=> SELECT u.User_ID, u.Name,
       ROUND(SUM(EXTRACT(EPOCH FROM str.Duration) / 3600), 2) AS Hours_Watched
FROM Ott_User u
JOIN Streams str ON u.User_ID = str.User_ID
WHERE u.Subscription_type = 'Premium'
  AND str.Stream_date >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY u.User_ID, u.Name
HAVING SUM(EXTRACT(EPOCH FROM str.Duration) / 3600) > 5;


3. Get the total revenue generated by each genre in 2024.
=> SELECT cg.Genre, SUM(s.Revenue) AS Total_Revenue
   FROM Content_Genre cg
   JOIN Statistics s ON cg.Content_ID = s.Content_ID
   JOIN Content c ON c.Content_ID = cg.Content_ID
   WHERE EXTRACT(YEAR FROM c.Release_Date) = 2024
   GROUP BY cg.Genre
   ORDER BY Total_Revenue DESC;

4. Find actors who have worked in both 'Drama' and 'Action' genres.
=> SELECT a.Name
   FROM Actor a
   JOIN Starred_in si ON a.Actor_ID = si.Actor_ID
   JOIN Content_Genre cg ON si.Content_ID = cg.Content_ID
   WHERE cg.Genre IN ('Drama', 'Action')
   GROUP BY a.Actor_ID, a.Name
   HAVING COUNT(DISTINCT cg.Genre) = 2;

5. List TV series whose average episode rating is above 8.5.
=> SELECT c.Title, ROUND(AVG(e.Rating)::numeric, 2) AS Avg_Rating
   FROM TV_Series ts
   JOIN Content c ON ts.Content_ID = c.Content_ID
   JOIN Episode e ON e.Content_ID = ts.Content_ID
   GROUP BY c.Title
   HAVING AVG(e.Rating) > 8.5;

6. Get the city-wise count of premium subscribers.
=> SELECT City, COUNT(*) AS Premium_Users
   FROM Ott_User
   WHERE Subscription_type = 'Premium'
   GROUP BY City
   ORDER BY Premium_Users DESC;

7. Identify contents that generated more than double their budget.
=> SELECT c.Title, s.Revenue, s.Budget
   FROM Content c
   JOIN Statistics s ON c.Content_ID = s.Content_ID
   WHERE c.Content_Type = 'Movie'
     AND s.Revenue >= 2 * s.Budget;

8. Find the average watch time per user in the last quarter.
=> SELECT 
    sub.User_ID, 
    sub.Name, 
    ROUND(sub.user_watch_time::numeric, 2) AS Watch_Hours
FROM (
    SELECT 
        u.User_ID, 
        u.Name,
        SUM(EXTRACT(EPOCH FROM str.Duration) / 3600) AS user_watch_time
    FROM Ott_User u
    JOIN Streams str ON u.User_ID = str.User_ID
    GROUP BY u.User_ID, u.Name
) sub;

9. Find the most popular subtitle language among premium users.
=> SELECT cs.Subtitle, COUNT(*) AS Usage_Count
   FROM Streams str
   JOIN Content_Subtitle cs ON str.Content_ID = cs.Content_ID
   JOIN Ott_User u ON str.User_ID = u.User_ID
   WHERE u.Subscription_type = 'Premium'
   GROUP BY cs.Subtitle
   ORDER BY Usage_Count DESC
   LIMIT 1;

10. List the top 3 production companies by revenue in 2024.
=> SELECT p.Name AS Production_Company, SUM(s.Revenue) AS Total_Revenue
   FROM Production p
   JOIN Content c ON p.Production_ID = c.Production_ID
   JOIN Statistics s ON c.Content_ID = s.Content_ID
   WHERE EXTRACT(YEAR FROM c.Release_Date) = 2024
   GROUP BY p.Name
   ORDER BY Total_Revenue DESC
   LIMIT 3;

11. Show the growth in total streams month by month in 2025.
=> SELECT TO_CHAR(str.Stream_date, 'Month') AS Month_Name, COUNT(*) AS Total_Streams
FROM Streams str
WHERE EXTRACT(YEAR FROM str.Stream_date) = 2025
GROUP BY Month_Name
ORDER BY MIN(str.Stream_date);


12. Get all upcoming movies featuring 'Ranbir Kapoor'.
=> SELECT c.Title, c.Release_Date
   FROM Coming_soon cs
   JOIN Content c ON cs.Content_ID = c.Content_ID
   JOIN starred_in si ON si.Content_ID = c.Content_ID
   JOIN Actor a ON si.Actor_ID = a.Actor_ID
   WHERE a.Name = 'Ranbir Kapoor';

13. Find the most profitable content (highest revenue minus budget).
=> SELECT c.Title, (s.Revenue - s.Budget) AS Profit
   FROM Content c
   JOIN Statistics s ON c.Content_ID = s.Content_ID
   ORDER BY Profit DESC
   LIMIT 1;

14. List the top 5 cities by total watch time in the last 6 months.
=> SELECT u.City, ROUND(SUM(EXTRACT(EPOCH FROM str.Duration)/3600), 2) AS Hours_Watched
   FROM Streams str
   JOIN Ott_User u ON str.User_ID = u.User_ID
   WHERE str.Stream_date >= CURRENT_DATE - INTERVAL '6 months'
   GROUP BY u.City
   ORDER BY Hours_Watched DESC
   LIMIT 5;

15. How many distinct users watched each genre in 2024.
=> SELECT cg.Genre, COUNT(DISTINCT str.User_ID) AS Unique_Users
   FROM Streams str
   JOIN Content_Genre cg ON str.Content_ID = cg.Content_ID
   JOIN Content c ON cg.Content_ID = c.Content_ID
   WHERE EXTRACT(YEAR FROM str.Stream_date) = 2024
   GROUP BY cg.Genre
   ORDER BY Unique_Users DESC;
